{====================================================================
  指标名称: MACD0轴信号副图 (配合双均线择时使用)
  适用平台: 同花顺 (THS) - 期货
  指标类型: 副图
  用途: 直观显示 MACD 的 0轴/DIFF/DEA, 并标出双均线在0轴
        上/下方的金叉死叉信号位置, 便于核对主图择时。
====================================================================}

N1:=10;
N2:=20;
SHORT:=12;
LONG:=26;
MID:=9;

{------------ MACD ------------}
DIF:EMA(CLOSE,SHORT)-EMA(CLOSE,LONG),COLORWHITE;
DEA:EMA(DIF,MID),COLORYELLOW;
MACD:(DIF-DEA)*2,COLORSTICK;
0轴:0,COLORGRAY,DOTLINE;

{------------ 双均线交叉(主图同源) ------------}
MA10:=MA(CLOSE,N1);
MA20:=MA(CLOSE,N2);
金叉:=CROSS(MA10,MA20);
死叉:=CROSS(MA20,MA10);

做多:=金叉 AND DIF>0 AND DEA>0;
做空:=死叉 AND DIF<0 AND DEA<0;

{------------ 在0轴附近标注信号 ------------}
DRAWICON(做多,0,1);
DRAWICON(做空,0,2);
