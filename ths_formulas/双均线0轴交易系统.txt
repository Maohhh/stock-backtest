{====================================================================
  名称: 双均线0轴期货交易系统 (可回测/自动交易信号)
  平台: 同花顺 (THS) - 期货 - 交易系统(TS)
  说明: 与"双均线0轴期货指标"同源, 输出 ENTERLONG/ENTERSHORT
        等交易系统关键字, 可直接用于同花顺期货交易系统回测。

  开平逻辑:
    开多: 0轴上方 MA10 金叉 MA20
    开空: 0轴下方 MA10 死叉 MA20
    平多: 出现死叉(趋势反转)
    平空: 出现金叉(趋势反转)
====================================================================}

N1:=10;
N2:=20;
BASE:=60;

基准:=MA(CLOSE,BASE);
偏离:=IF(基准>0,(CLOSE-基准)/基准*100,0);

MA10:=MA(偏离,N1);
MA20:=MA(偏离,N2);

金叉:=CROSS(MA10,MA20);
死叉:=CROSS(MA20,MA10);

{ ---- 交易系统信号 ---- }
ENTERLONG:金叉 AND MA10>0 AND MA20>0;   { 0轴上方金叉 -> 开多 }
EXITLONG:死叉;                          { 死叉 -> 平多 }

ENTERSHORT:死叉 AND MA10<0 AND MA20<0;  { 0轴下方死叉 -> 开空 }
EXITSHORT:金叉;                         { 金叉 -> 平空 }
