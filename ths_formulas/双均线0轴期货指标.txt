{====================================================================
  指标名称: 双均线0轴期货指标 (MA10/MA20 + 0轴金叉死叉)
  适用平台: 同花顺 (THS) - 期货
  指标类型: 副图 / 可挂接交易系统
  作者备注: maohanxxxxx@gmail.com

  设计逻辑:
    价格均线本身没有"0轴"概念(价格在几千点上下,不会穿0),
    因此先把价格转换成"相对长期基准的偏离百分比",该序列
    天然围绕 0 轴上下波动; 再对它取 MA10 与 MA20 两条均线。

    - 0轴上方 且 MA10 上穿 MA20 (金叉)  -> 做多
    - 0轴下方 且 MA10 下穿 MA20 (死叉)  -> 做空

    基准周期 BASE 默认 60, 可按品种/周期调整(参数化)。
====================================================================}

{------------ 参数(可在指标设置里调节) ------------}
N1:=10;          { 快线周期 }
N2:=20;          { 慢线周期 }
BASE:=60;        { 0轴基准周期, 长期趋势中枢 }

{------------ 0轴振荡基础序列 ------------}
基准:=MA(CLOSE,BASE);
偏离:=IF(基准>0,(CLOSE-基准)/基准*100,0);   { 相对基准的偏离%, 围绕0轴波动 }

{------------ 两条均线 ------------}
MA10:MA(偏离,N1),COLORWHITE,LINETHICK2;
MA20:MA(偏离,N2),COLORYELLOW,LINETHICK2;
0轴:0,COLORGRAY,DOTLINE;

{------------ 金叉 / 死叉 ------------}
金叉:=CROSS(MA10,MA20);
死叉:=CROSS(MA20,MA10);

{------------ 0轴过滤后的多空信号 ------------}
做多:=金叉 AND MA10>0 AND MA20>0;   { 0轴上方金叉 }
做空:=死叉 AND MA10<0 AND MA20<0;   { 0轴下方死叉 }

{------------ 信号绘制 ------------}
STICKLINE(做多,MA10,MA20,3,0),COLORRED;
STICKLINE(做空,MA10,MA20,3,0),COLORGREEN;
DRAWICON(做多,MA10,1);
DRAWICON(做空,MA10,2);
DRAWTEXT(做多,MA20,'做多'),COLORRED;
DRAWTEXT(做空,MA10,'做空'),COLORGREEN;
