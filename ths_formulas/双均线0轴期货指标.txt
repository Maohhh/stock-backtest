{====================================================================
  指标名称: 双均线+MACD0轴 期货择时指标
  适用平台: 同花顺 (THS) - 期货
  指标类型: 主图叠加 (画在K线主图上)
  作者备注: maohanxxxxx@gmail.com

  设计逻辑:
    方向用 MACD 的 0轴(DIFF/DEA 围绕的零轴)过滤,
    择时用 MA10 / MA20 的金叉死叉:

    - MACD 在 0轴上方(DIFF>0) 且 MA10 金叉 MA20  -> 做多
    - MACD 在 0轴下方(DIFF<0) 且 MA10 死叉 MA20  -> 做空

    即: 顺 MACD 趋势方向, 用双均线交叉择时入场。
====================================================================}

{------------ 参数 ------------}
N1:=10;          { 快线 MA 周期 }
N2:=20;          { 慢线 MA 周期 }
SHORT:=12;       { MACD 快speed EMA }
LONG:=26;        { MACD 慢speed EMA }
MID:=9;          { MACD DEA 周期 }

{------------ 双均线(主图) ------------}
MA10:MA(CLOSE,N1),COLORWHITE,LINETHICK2;
MA20:MA(CLOSE,N2),COLORYELLOW,LINETHICK2;

{------------ MACD: 提供0轴方向 ------------}
DIF:=EMA(CLOSE,SHORT)-EMA(CLOSE,LONG);
DEA:=EMA(DIF,MID);

{------------ 金叉 / 死叉 ------------}
金叉:=CROSS(MA10,MA20);
死叉:=CROSS(MA20,MA10);

{------------ 0轴过滤后的多空信号 ------------}
做多:=金叉 AND DIF>0;     { 0轴上方金叉 }
做空:=死叉 AND DIF<0;     { 0轴下方死叉 }

{------------ 信号绘制(主图) ------------}
DRAWICON(做多,LOW,1);
DRAWICON(做空,HIGH,2);
DRAWTEXT(做多,LOW,'做多'),COLORRED;
DRAWTEXT(做空,HIGH,'做空'),COLORGREEN;
